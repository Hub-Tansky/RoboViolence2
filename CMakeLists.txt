set(BV2_PLATFORM_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/engine/zeven/include/platform.h")

# Flags shared by every target: platform.h force-included, Windows 10/11 API level, sanitizers.
function(bv2_target_defaults target)
  if(MSVC)
    target_compile_options(${target} PRIVATE "/FI${BV2_PLATFORM_HEADER}" /utf-8)
    target_compile_definitions(${target} PRIVATE
      _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE NOMINMAX)
    if(BV2_SANITIZE)
      target_compile_options(${target} PRIVATE /fsanitize=address)
    endif()
  else()
    target_compile_options(${target} PRIVATE -include "${BV2_PLATFORM_HEADER}")
    if(BV2_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  if(APPLE)
    target_compile_definitions(${target} PRIVATE GL_SILENCE_DEPRECATION)
  endif()
  if(WIN32)
    target_compile_definitions(${target} PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  endif()
endfunction()

# Runtime layout: <build>/runtime/{bin, main -> content, bv2.db, master.db}
set(BV2_RUNTIME_DIR "${CMAKE_BINARY_DIR}/runtime")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${BV2_RUNTIME_DIR}")
foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${cfg} "${BV2_RUNTIME_DIR}")
endforeach()
file(MAKE_DIRECTORY "${BV2_RUNTIME_DIR}")
if(NOT EXISTS "${BV2_RUNTIME_DIR}/main")
  file(CREATE_LINK "${CMAKE_SOURCE_DIR}/content" "${BV2_RUNTIME_DIR}/main" SYMBOLIC RESULT bv2_link_result)
  if(NOT bv2_link_result STREQUAL "0" AND WIN32)
    # No symlink privilege: use a directory junction.
    file(TO_NATIVE_PATH "${BV2_RUNTIME_DIR}/main" bv2_link_native)
    file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}/content" bv2_target_native)
    execute_process(COMMAND cmd /c mklink /J "${bv2_link_native}" "${bv2_target_native}"
                    OUTPUT_QUIET RESULT_VARIABLE bv2_link_result)
  endif()
  if(NOT bv2_link_result STREQUAL "0")
    message(WARNING "Could not link ${BV2_RUNTIME_DIR}/main to content/. Link or copy it by hand.")
  endif()
endif()
